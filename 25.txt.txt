Q1. Calculator interface add/subtract
interface Calculator{int add(int a,int b);int subtract(int a,int b);}
class SimpleCalc implements Calculator{public int add(int a,int b){return a+b;}public int subtract(int a,int b){return a-b;}}
public class CalcDemo{public static void main(String[]x){SimpleCalc c=new SimpleCalc();System.out.println(c.add(10,5));System.out.println(c.subtract(10,5));}}
