Q1. Array sum and ascending sort
import java.util.*;
public class ArraySumSort {
 public static void main(String[] args) {
  Scanner sc=new Scanner(System.in); System.out.print("How many elements? "); int n=sc.nextInt();
  int[] a=new int[n]; int sum=0;
  for(int i=0;i<n;i++){a[i]=sc.nextInt();sum+=a[i];}
  Arrays.sort(a); System.out.println("Sum = "+sum);
  System.out.println("Ascending = "+Arrays.toString(a));
 }
}
