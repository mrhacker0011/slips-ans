Q1. JavaFX color buttons
import javafx.application.Application;import javafx.scene.Scene;import javafx.scene.control.*;import javafx.scene.layout.*;import javafx.stage.Stage;
public class ColorButtonsFX extends Application{public void start(Stage s){Pane p=new Pane();HBox h=new HBox(10);for(String c:new String[]{"red","green","blue"}){Button b=new Button(c);b.setOnAction(e->p.setStyle("-fx-background-color: "+c+";"));h.getChildren().add(b);}p.getChildren().add(h);s.setScene(new Scene(p,350,200));s.show();}public static void main(String[]x){launch(x);}}
