Q1. Read text file, reverse text and change case
import java.nio.file.*;import java.util.*;
public class ReverseCaseFile{
 public static void main(String[]args)throws Exception{Scanner s=new Scanner(System.in);System.out.print("File path: ");String p=s.nextLine();String t=Files.readString(Path.of(p));StringBuilder rev=new StringBuilder(t).reverse();StringBuilder out=new StringBuilder();for(char c:rev.toString().toCharArray())out.append(Character.isUpperCase(c)?Character.toLowerCase(c):Character.toUpperCase(c));System.out.println(out);}
}
