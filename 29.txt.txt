Q1. Static method: zero exception and prime check
import java.util.*;
class ZeroNumberException extends Exception{ZeroNumberException(String m){super(m);}}
public class PrimeStatic{static void check(int n)throws ZeroNumberException{if(n==0)throw new ZeroNumberException("Number is 0");if(n<2){System.out.println("Not prime");return;}for(int i=2;i<=Math.sqrt(n);i++)if(n%i==0){System.out.println("Not prime");return;}System.out.println("Prime");}
public static void main(String[]x){Scanner s=new Scanner(System.in);try{check(s.nextInt());}catch(Exception e){System.out.println(e.getMessage());}}}
