Q1. Mouse moved/click coordinates in JavaFX
import javafx.application.Application;import javafx.scene.Scene;import javafx.scene.control.*;import javafx.scene.layout.*;import javafx.stage.Stage;
public class MouseFX extends Application{public void start(Stage s){TextField out=new TextField();Pane p=new Pane();p.setPrefSize(400,250);p.setOnMouseMoved(e->s.setTitle("Moved: "+e.getX()+", "+e.getY()));p.setOnMouseClicked(e->out.setText("Clicked: "+e.getX()+", "+e.getY()));VBox root=new VBox(8,p,out);s.setScene(new Scene(root,420,320));s.show();}public static void main(String[]x){launch(x);}}
