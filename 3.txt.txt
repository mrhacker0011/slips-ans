Q1. String package Con and Comp
// Save in folder stringops/
// Con.java: package stringops; public class Con { public void join(String a,String b){System.out.println("Concatenated: "+a+b);} }
// Comp.java: package stringops; public class Comp { public void compare(String a,String b){System.out.println(a.equals(b)?"Strings are equal":"Strings are not equal");} }
// Main.java (outside package):
import stringops.Con; import stringops.Comp;
public class Main { public static void main(String[] args){new Con().join("Core ","Java");new Comp().compare("Java","java");} }
// Compile: javac stringops/*.java Main.java ; run: java Main
