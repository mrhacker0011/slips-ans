Q1. Armstrong numbers in a range
import java.util.*;

public class ArmstrongRange {

 static boolean armstrong(int n){
  int x=n,d=String.valueOf(n).length(),sum=0;
   do{
  int r=x%10;
  sum+=(int)Math.pow(r,d);
  x/=10;

}
while(x>0);
  return sum==n;

}

 public static void main(String[] args){
  Scanner s=new Scanner(System.in);
  System.out.print("Start and end: ");
  int a=s.nextInt(),b=s.nextInt();
  for(int i=Math.max(0,a);
  i<=b;
  i++)if(armstrong(i))System.out.print(i+" ");

}

}
