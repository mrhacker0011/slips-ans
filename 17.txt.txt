Q1. JavaFX change background with Ctrl+Alt+O
import javafx.application.Application;import javafx.scene.Scene;import javafx.scene.input.KeyCode;import javafx.scene.layout.*;import javafx.stage.Stage;
public class KeyColorFX extends Application{public void start(Stage s){Pane p=new Pane();Scene sc=new Scene(p,400,250);sc.setOnKeyPressed(e->{if(e.isControlDown()&&e.isAltDown()&&e.getCode()==KeyCode.O)p.setStyle("-fx-background-color: orange;");});s.setScene(sc);s.setTitle("Press Ctrl+Alt+O");s.show();p.requestFocus();}public static void main(String[]x){launch(x);}}
